# Start Redis, Celery Worker, and FastAPI Backend in a single command (Standard Mode)
source venv/bin/activate && (redis-server & celery -A celery_app worker --loglevel=info & uvicorn server:app --host 0.0.0.0 --port 8001 --reload)

# Start Backend in Apify DOCKER MODE (Phase 7)
source backend/venv/bin/activate && export USE_DOCKER_EXECUTOR=true && export DOCKER_HOST="unix:///Users/gokul/.colima/default/docker.sock" && (redis-server & cd backend && celery -A celery_app worker --loglevel=info & uvicorn server:app --host 0.0.0.0 --port 8001 --reload)

# Run this if you need to kill everything (Backend, Redis, Celery, and React/Vite/NPM dev servers)
